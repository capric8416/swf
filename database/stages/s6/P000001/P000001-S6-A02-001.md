```sql
-- 技术选型理由：简单轻量，无需额外服务，与 PostgreSQL 深度集成

-- 监控扩展
CREATE EXTENSION IF NOT EXISTS pg_stat_statements;
```

**定时调度器选型说明**：

| 选型 | 方案 | 理由 |
|------|------|------|
| **原设计** | Celery Beat | 功能强大，但需要额外服务 |
| **现设计** | pg_cron | 简单轻量，与 PostgreSQL 深度集成，运维成本低 |
| **适用场景** | 简单定时任务 | 每天凌晨 2 点数据同步、每小时统计计算 |
| **未来扩展** | 如需复杂调度（任务链、重试、分布式），可迁移回 Celery Beat |

```sql
-- 创建用户表
CREATE TABLE users (
    id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    username VARCHAR(50) NOT NULL UNIQUE,
    email VARCHAR(100) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    department VARCHAR(50) NOT NULL,
    role_id INTEGER,
    is_active BOOLEAN DEFAULT TRUE NOT NULL,
    last_login_at TIMESTAMP,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,
    
    -- 外键约束
    CONSTRAINT fk_users_role
        FOREIGN KEY (role_id) REFERENCES roles(id)
        ON DELETE SET NULL,
    
    -- 检查约束
    CONSTRAINT chk_users_email_format
        CHECK (email ~* '^[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}$'),
    CONSTRAINT chk_users_username_length
        CHECK (LENGTH(username) >= 3 AND LENGTH(username) <= 50)

CREATE INDEX idx_users_role_id ON users(role_id);
CREATE INDEX idx_users_is_active ON users(is_active);
CREATE INDEX idx_users_created_at ON users(created_at);

-- 触发器：自动更新 updated_at

    BEFORE UPDATE ON users
    FOR EACH ROW
    EXECUTE FUNCTION update_updated_at_column();
```

#### 3.6.1 分区创建函数

```sql
-- 创建分区的存储过程
CREATE OR REPLACE FUNCTION create_monthly_partition(
    table_name TEXT,
    year INTEGER,
    month INTEGER
) RETURNS VOID AS $$
DECLARE
    partition_name TEXT;
    start_date DATE;
    end_date DATE;
BEGIN
    partition_name := table_name || '_' || year || '_' || LPAD(month::TEXT, 2, '0');
    start_date := MAKE_DATE(year, month, 1);
    end_date := start_date + INTERVAL '1 month';
    
    EXECUTE format(
        'CREATE TABLE IF NOT EXISTS %I PARTITION OF %I
         FOR VALUES FROM (%L) TO (%L)',
        partition_name, table_name, start_date, end_date
    );
    
    RAISE NOTICE 'Created partition: %', partition_name;
END;
$$ LANGUAGE plpgsql;
```

### 4.2 审计日志触发器函数

**审计范围**：所有写操作（INSERT/UPDATE/DELETE）

**审计对象**：
- 用户创建、更新、删除
- 项目创建、更新、删除
- 配置创建、更新、删除
- 权限变更
- 数据同步操作
- 登录/登出
- Token 刷新

**不记录的操作**：
- 查询操作（SELECT）
- 统计数据访问

```sql
-- 创建审计日志表
CREATE TABLE audit_logs (
    id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    table_name VARCHAR(50) NOT NULL,
    record_id INTEGER NOT NULL,
    action VARCHAR(20) NOT NULL,
    old_data JSONB,
    new_data JSONB,
    user_id INTEGER,
    ip_address INET,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL
);

-- 索引
CREATE INDEX idx_audit_logs_table ON audit_logs(table_name);
CREATE INDEX idx_audit_logs_record ON audit_logs(record_id);
CREATE INDEX idx_audit_logs_user ON audit_logs(user_id);
CREATE INDEX idx_audit_logs_created ON audit_logs(created_at);

-- 创建审计日志触发器函数
CREATE OR REPLACE FUNCTION log_audit_changes()
RETURNS TRIGGER AS $$
BEGIN
    IF TG_OP = 'INSERT' THEN
        INSERT INTO audit_logs (table_name, record_id, action, new_data, user_id)
        VALUES (TG_TABLE_NAME, NEW.id, 'INSERT', to_jsonb(NEW), 
                current_setting('app.current_user_id', TRUE)::INTEGER);
        RETURN NEW;
    ELSIF TG_OP = 'UPDATE' THEN
        INSERT INTO audit_logs (table_name, record_id, action, old_data, new_data, user_id)
        VALUES (TG_TABLE_NAME, NEW.id, 'UPDATE', to_jsonb(OLD), to_jsonb(NEW),
                current_setting('app.current_user_id', TRUE)::INTEGER);
        RETURN NEW;
    ELSIF TG_OP = 'DELETE' THEN
        INSERT INTO audit_logs (table_name, record_id, action, old_data, user_id)
        VALUES (TG_TABLE_NAME, OLD.id, 'DELETE', to_jsonb(OLD),
                current_setting('app.current_user_id', TRUE)::INTEGER);
        RETURN OLD;
    END IF;
    RETURN NULL;
END;
$$ LANGUAGE plpgsql;

COMMENT ON FUNCTION log_audit_changes() IS '审计日志触发器函数 - 记录数据变更';
```

### 5.2 统计相关存储过程

#### 5.2.1 计算项目统计数据

```sql
-- 计算项目统计数据
CREATE OR REPLACE FUNCTION calculate_project_stats(
    p_project_id INTEGER,
    p_start_date DATE DEFAULT NULL,
    p_end_date DATE DEFAULT NULL
) RETURNS JSONB AS $$
DECLARE
    v_stats JSONB;
BEGIN
    SELECT jsonb_build_object(
        'code_stats', jsonb_build_object(
            'total_commits', COUNT(cc.id),
            'total_additions', COALESCE(SUM(cc.additions), 0),
            'total_deletions', COALESCE(SUM(cc.deletions), 0),
            'net_additions', COALESCE(SUM(cc.additions - cc.deletions), 0),
            'ai_commits', COUNT(*) FILTER (WHERE cc.is_ai_generated),
            'ai_adoption_rate', CASE 
                WHEN COUNT(cc.id) > 0 
                THEN ROUND(COUNT(*) FILTER (WHERE cc.is_ai_generated)::NUMERIC / COUNT(cc.id)::NUMERIC, 4)
                ELSE 0
            END
        ),
        'token_stats', jsonb_build_object(
            'total_tokens', COALESCE(SUM(tu.token_count), 0),
            'total_calls', COALESCE(SUM(tu.api_calls), 0)
        ),
        'bug_stats', jsonb_build_object(
            'total_bugs', COUNT(DISTINCT br.id),
            'open_bugs', COUNT(DISTINCT br.id) FILTER (WHERE br.status NOT IN ('resolved', 'closed', 'rejected')),
            'resolved_bugs', COUNT(DISTINCT br.id) FILTER (WHERE br.status IN ('resolved', 'closed'))
        ),
        'team_stats', jsonb_build_object(
            'member_count', COUNT(DISTINCT pm.user_id),
            'contributor_count', COUNT(DISTINCT cc.user_id)
        )
    ) INTO v_stats
    FROM projects p
    LEFT JOIN code_commits cc ON cc.project_id = p.id 
        AND (p_start_date IS NULL OR cc.commit_time >= p_start_date)
        AND (p_end_date IS NULL OR cc.commit_time <= p_end_date)
    LEFT JOIN token_usage tu ON tu.project_id = p.id
        AND (p_start_date IS NULL OR tu.usage_date >= p_start_date)
        AND (p_end_date IS NULL OR tu.usage_date <= p_end_date)
    LEFT JOIN bug_records br ON br.project_id = p.id
        AND (p_start_date IS NULL OR br.created_at >= p_start_date)
        AND (p_end_date IS NULL OR br.created_at <= p_end_date)
    LEFT JOIN project_members pm ON pm.project_id = p.id AND pm.left_at IS NULL
    WHERE p.id = p_project_id
    GROUP BY p.id;
    
    RETURN COALESCE(v_stats, '{}'::jsonb);
END;
$$ LANGUAGE plpgsql;

COMMENT ON FUNCTION calculate_project_stats() IS '计算项目统计数据';
```

#### 5.2.2 创建统计快照

```sql
-- 创建每日统计快照
CREATE OR REPLACE FUNCTION create_daily_snapshot(
    p_snapshot_date DATE DEFAULT CURRENT_DATE
) RETURNS INTEGER AS $$
DECLARE
    v_count INTEGER := 0;
BEGIN
    -- 全局快照
    INSERT INTO stats_snapshots (snapshot_type, snapshot_date, metrics)
    SELECT 
        'global',
        p_snapshot_date,
        jsonb_build_object(
            'total_users', (SELECT COUNT(*) FROM users WHERE is_active),
            'total_projects', (SELECT COUNT(*) FROM projects WHERE status = 'active'),
            'total_commits', (SELECT COUNT(*) FROM code_commits WHERE commit_time::DATE = p_snapshot_date),
            'total_tokens', (SELECT COALESCE(SUM(token_count), 0) FROM token_usage WHERE usage_date = p_snapshot_date),
            'total_bugs', (SELECT COUNT(*) FROM bug_records WHERE created_at::DATE = p_snapshot_date)
        )
    ON CONFLICT (snapshot_type, snapshot_date, project_id, user_id) 
    DO UPDATE SET metrics = EXCLUDED.metrics;
    
    v_count := v_count + 1;
    
    -- 项目快照
    INSERT INTO stats_snapshots (snapshot_type, snapshot_date, project_id, metrics)
    SELECT 
        'project',
        p_snapshot_date,
        p.id,
        calculate_project_stats(p.id, p_snapshot_date, p_snapshot_date)
    FROM projects p
    WHERE p.status = 'active';
    
    v_count := v_count + (SELECT COUNT(*) FROM projects WHERE status = 'active');
    
    RAISE NOTICE 'Created % snapshots for %', v_count, p_snapshot_date;
    RETURN v_count;
END;
$$ LANGUAGE plpgsql;

COMMENT ON FUNCTION create_daily_snapshot() IS '创建每日统计快照';
```

```sql
    SUM(deletions) AS total_deletions,
    COUNT(*) FILTER (WHERE is_ai_generated) AS ai_commits,
    COUNT(DISTINCT user_id) AS contributor_count
FROM code_commits
GROUP BY project_id, DATE_TRUNC('day', commit_time)::DATE
WITH DATA;

-- 索引
CREATE INDEX mv_idx_project_code_stats_project ON mv_project_code_stats(project_id);
CREATE INDEX mv_idx_project_code_stats_date ON mv_project_code_stats(stat_date);
CREATE UNIQUE INDEX mv_idx_project_code_stats_unique 
    ON mv_project_code_stats(project_id, stat_date);

-- 刷新物化视图的函数
CREATE OR REPLACE FUNCTION refresh_project_code_stats()
RETURNS VOID AS $$
BEGIN
    REFRESH MATERIALIZED VIEW CONCURRENTLY mv_project_code_stats;
    RAISE NOTICE 'Materialized view mv_project_code_stats refreshed';
END;
$$ LANGUAGE plpgsql;
```
